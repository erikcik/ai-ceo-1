# The denylist safety-gate.sh enforces. One rule per line, TAB-separated:
#
#   CATEGORY <TAB> extended-regex (case-insensitive) <TAB> why it is blocked
#
# CATEGORY is one of MONEY, PUBLISH, IRREVERSIBLE. Lines starting with # are
# comments. Edit this file to fit your deployment; safety-gate.sh needs no
# changes. Every match is logged to PAUSED_ACTIONS.md and the action is denied.
#
# NEVER put a literal tab inside a regex -- TAB is the field separator, so a tab
# in the pattern splits the rule and breaks it. Use \s for whitespace. An
# unparseable rule makes the gate fail CLOSED (every call is blocked until you
# fix it), because a safety gate that quietly drops rules is worse than none.
#
# The regex is matched against a narrow, purpose-built haystack -- the Bash
# command, the target file path, the request URL, and the tool name -- never
# against file contents, so writing prose about `git push` does not trip it.

# --- MONEY: anything that can move real money -------------------------------
MONEY	\bstripe\b	Stripe CLI/API can charge a card
MONEY	\b(paypal|braintree|adyen|squareup)\b	payment processor CLI/API
MONEY	\b(coinbase|binance)\b|\bledger[\s-]?live\b	crypto exchange / wallet
MONEY	(purchase|checkout|billing|invoice|payout|subscribe|top_?up)	purchase or billing operation
MONEY	buy_credits|setup_payment|confirm_billing	credit purchase tool
MONEY	\baws\b.*\b(ec2\s+run-instances|rds\s+create)	provisions billable cloud resources
MONEY	\b(gcloud|az)\b.*\b(create|deploy)\b.*\b(instance|cluster|vm)\b	provisions billable cloud resources

# --- PUBLISH: anything visible outside this machine -------------------------
PUBLISH	\bgit\s+push\b	publishes commits to a remote
PUBLISH	\bgh\s+(pr|issue|release|gist)\s+(create|edit|comment)	creates public GitHub content
PUBLISH	\b(npm|cargo)\s+publish\b|\btwine\s+upload\b	publishes a package
PUBLISH	\bcurl\b.*\s-X\s*(POST|PUT|PATCH|DELETE)	sends data to an external service
PUBLISH	\bcurl\b.*(hooks\.slack\.com|discord(app)?\.com/api/webhooks|api\.telegram\.org)	posts to a chat webhook
PUBLISH	\b(sendmail|mailx|mutt)\b|\bmail\s+-s\b	sends email
PUBLISH	send_message|create_draft|^mcp__.*__reply$|__forward$	sends or drafts an external message
PUBLISH	tiktok_publish|publish_website|deploy_website|participate_in_contest	posts publicly
PUBLISH	\b(vercel|netlify|fly|wrangler)\s+deploy\b	deploys to a public URL
PUBLISH	share_file|notion-create-comment	shares content with other people

# --- IRREVERSIBLE: real-world actions with no undo --------------------------
IRREVERSIBLE	\brm\s+(-\S*[rf]\S*\s+)+\S	recursive or forced delete
IRREVERSIBLE	\bgit\s+push\b.*(--force|-f\b)	rewrites published history
IRREVERSIBLE	\bgit\s+(reset\s+--hard|clean\s+-\S*f)	discards uncommitted work irrecoverably
IRREVERSIBLE	\bDROP\s+(TABLE|DATABASE|SCHEMA)\b|\bTRUNCATE\s+TABLE\b	destroys stored data
IRREVERSIBLE	\bterraform\s+(apply|destroy)\b	changes real infrastructure
IRREVERSIBLE	\bkubectl\s+delete\b|\bhelm\s+(uninstall|delete)\b	deletes running workloads
IRREVERSIBLE	\b(aws|gcloud|az)\b.*\bdelete\b|\baws\s+s3\s+rb\b	deletes cloud resources
IRREVERSIBLE	trash_file|trash_message|trash_thread|delete_event|notion-.*delete	deletes a record in an external system
IRREVERSIBLE	\bdd\s+if=|\bmkfs\b|\bdiskutil\s+erase	destroys a disk or volume
IRREVERSIBLE	\bcrontab\s+-r\b|\blaunchctl\s+(unload|bootout)\b	removes scheduled work outside this repo
